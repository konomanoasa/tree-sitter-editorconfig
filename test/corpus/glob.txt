==================
Nested alternatives and numeric ranges preserve ownership
==================
[{a,{1..3}}]

---

(document
  (section
    header: (section_header
      opening: (section_open)
      name: (pattern
        (alternation
          opening: (brace_open)
          (pattern (glob_literal))
          (alternative_separator)
          (pattern
            (numeric_range
              opening: (brace_open)
              lower: (integer)
              (range_separator)
              upper: (integer)
              closing: (brace_close)))
          closing: (brace_close)))
      closing: (section_close)
      (line_ending))))
